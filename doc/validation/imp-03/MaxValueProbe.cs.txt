
var original = SceneManager.GetActiveScene();
if (string.IsNullOrEmpty(original.path)) throw new InvalidOperationException("A saved original scene is required for this additive probe.");
int originalCount = SceneManager.sceneCount;
var probeScene = EditorSceneManager.NewScene(NewSceneSetup.EmptyScene, NewSceneMode.Additive);
SceneManager.SetActiveScene(probeScene);
var checks = new List<string>();
try
{
    foreach (var mode in new[] { SceneRootMode.SceneOwned, SceneRootMode.Singleton })
    {
        var owner = new GameObject("Imp03MaxValueProbe");
        owner.SetActive(false);
        try
        {
            Exception attachFailure = null;
            try { SceneRootSetup.Attach(owner, mode, null, false, float.MaxValue); }
            catch (Exception exception) { attachFailure = exception; }
            if (!(attachFailure is ArgumentOutOfRangeException) ||
                ((ArgumentOutOfRangeException)attachFailure).ParamName != "shutdownTimeoutSeconds" ||
                owner.GetComponent<SceneOwnedRoot>() != null || owner.GetComponent<SingletonSceneRoot>() != null)
                throw new InvalidOperationException("Attach did not reject before mutation: " + mode);
            checks.Add(mode + ".Attach:ArgumentOutOfRangeException/hostCount0");
            var host = SceneRootSetup.Attach(owner, mode, null, false, 0.25f);
            Exception configureFailure = null;
            try
            {
                if (host is SceneOwnedRoot sceneHost) sceneHost.Configure(null, true, float.MaxValue);
                else ((SingletonSceneRoot)host).Configure(null, true, float.MaxValue);
            }
            catch (Exception exception) { configureFailure = exception; }
            using (var serialized = new SerializedObject((UnityEngine.Object)host))
            {
                if (!(configureFailure is ArgumentOutOfRangeException) ||
                    ((ArgumentOutOfRangeException)configureFailure).ParamName != "shutdownTimeoutSeconds" ||
                    serialized.FindProperty("_shutdownTimeoutSeconds").floatValue != 0.25f ||
                    serialized.FindProperty("_persistAcrossScenes").boolValue)
                    throw new InvalidOperationException("Configure mutated existing settings: " + mode);
            }
            checks.Add(mode + ".Configure:ArgumentOutOfRangeException/settingsPreserved");
        }
        finally { UnityEngine.Object.DestroyImmediate(owner); }
    }
}
finally
{
    SceneManager.SetActiveScene(original);
    EditorSceneManager.CloseScene(probeScene, true);
}
if (SceneManager.sceneCount != originalCount || SceneManager.GetActiveScene() != original)
    throw new InvalidOperationException("Probe scene restoration failed.");
return string.Join("\n", checks);
