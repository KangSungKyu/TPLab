using System;
using System.Collections.Generic;
using System.Linq;
using System.Reflection;
using NUnit.Framework;
using TPLab.Core.SceneManagement;
using TPLab.Core.Tests;
using UnityEngine.SceneManagement;

public static class Imp05OptimizedProbe
{
    public static bool BooleanSink;
    public static int CountSink;
    public static IReadOnlyList<Scene> OwnedSink;
    public static IReadOnlyList<SceneRegistration> RegisteredSink;
    public struct Row
    {
        public string Component;
        public int Calls;
        public long Events;
        public long PositiveControlEvents;
        public long NegativeControlEvents;
        public bool FrameBytesAvailable;
        public long GlobalFrameByteDelta;
    }

    public static object Run(GameSceneManager manager)
    {
        Assert.That(manager.CanProceed, Is.True);
        var type = typeof(GameSceneManager);
        const BindingFlags instancePrivate = BindingFlags.Instance | BindingFlags.NonPublic;
        const BindingFlags staticPrivate = BindingFlags.Static | BindingFlags.NonPublic;
        var registered = (Func<bool>)Delegate.CreateDelegate(typeof(Func<bool>), manager,
            type.GetMethod("AreRegisteredOwnersPrepared", instancePrivate));
        var expectedInventory = (Func<bool>)Delegate.CreateDelegate(typeof(Func<bool>), manager,
            type.GetMethod("InventoryMatchesExpectedGameScenes", instancePrivate));
        var arrayInventory = (Func<Scene[], bool>)Delegate.CreateDelegate(typeof(Func<Scene[], bool>),
            type.GetMethod("InventoryMatches", staticPrivate));
        var retained = (Scene[])type.GetField("_retainedScenes", instancePrivate).GetValue(manager);
        var expected = retained.Concat(manager.OwnedScenes).ToArray();
        Assert.That(type.GetMethod("ExpectedGameScenes", instancePrivate), Is.Null);
        var owned = new List<Scene>(8);
        var registrations = new List<SceneRegistration>(8);
        string[] names = { "actual registered index loop", "actual expected membership + dynamic inventory",
            "actual array InventoryMatches with prebuilt expected", "actual CanProceed getter",
            "actual OwnedSceneCount", "actual GetOwnedScenes", "actual GetRegisteredScenes",
            "legacy OwnedScenes snapshot", "legacy RegisteredScenes snapshot" };
        int[] calls = { 100, 100, 100, 100, 1000, 1000, 1000, 100, 100 };
        Action[] workloads = {
            () => { bool value = true; for (int i=0;i<100;i++) value &= registered(); BooleanSink = value; },
            () => { bool value = true; for (int i=0;i<100;i++) value &= expectedInventory(); BooleanSink = value; },
            () => { bool value = true; for (int i=0;i<100;i++) value &= arrayInventory(expected); BooleanSink = value; },
            () => { bool value = true; for (int i=0;i<100;i++) value &= manager.CanProceed; BooleanSink = value; },
            () => { int value = 0; for (int i=0;i<1000;i++) value += manager.OwnedSceneCount; CountSink = value; },
            () => { for (int i=0;i<1000;i++) manager.GetOwnedScenes(owned); },
            () => { for (int i=0;i<1000;i++) manager.GetRegisteredScenes(registrations); },
            () => { for (int i=0;i<100;i++) OwnedSink = manager.OwnedScenes; },
            () => { for (int i=0;i<100;i++) RegisteredSink = manager.RegisteredScenes; }
        };
        var rows = new Row[workloads.Length];
        for (int i=0;i<workloads.Length;i++)
        {
            workloads[i]();
            var measurement = GcAllocationMeasurement.Measure(workloads[i]);
            rows[i] = new Row { Component=names[i], Calls=calls[i], Events=measurement.EventCount,
                PositiveControlEvents=measurement.PositiveControlEventCount,
                NegativeControlEvents=measurement.NegativeControlEventCount,
                FrameBytesAvailable=measurement.FrameBytesAvailable, GlobalFrameByteDelta=measurement.FrameByteDelta };
            if (i<7) Assert.That(measurement.EventCount, Is.Zero);
            else Assert.That(measurement.EventCount, Is.GreaterThan(0));
        }
        return new { completed=true, oldExpectedArrayBuilderRemoved=true, primaryOnly=true,
            observations=rows, ownedCapacity=owned.Capacity, registeredCapacity=registrations.Capacity,
            limitations="Actual bound optimized private methods and public APIs; all binding/warm-up/buffer allocation outside measured windows. Expected union and dynamic membership now share one method, so former ExpectedGameScenes has no separate after row. Global frame bytes are diagnostics. Nested3scene combined buffer parity/1000call zero validation is in the permanent test fixture. Player/IL2CPP NotRun." };
    }
}
