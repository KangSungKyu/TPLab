using System;
using System.Collections.Generic;
using System.Linq;
using System.Reflection;
using System.Reflection.Emit;
using NUnit.Framework;
using TPLab.Core.SceneManagement;
using TPLab.Core.Tests;
using UnityEngine.SceneManagement;

// Transient CLI probe only. Caller supplies its actual Ready manager and owns its cleanup.
public static class Imp05ComponentProbe
{
    public const int CallsPerComponent = 100;
    public static readonly ComponentResult[] Results = new ComponentResult[4];
    public static Scene[] ExpectedSink;
    public static Scene[] InventoryExpected;
    public static bool BooleanSink;
    public static string GetterIlHex;
    public static bool OriginalAllSequenceVerified;

    public struct ComponentResult
    {
        public string Component;
        public int Calls;
        public long Events;
        public long PositiveControlEvents;
        public long NegativeControlEvents;
        public bool FrameBytesAvailable;
        public long GlobalFrameByteDelta;
    }

    private struct Instruction
    {
        public OpCode Code;
        public MethodBase Method;
    }

    public static object Run(GameSceneManager manager)
    {
        if (manager == null) throw new ArgumentNullException(nameof(manager));
        Assert.That(manager.State, Is.EqualTo(SceneTransitionState.Ready));
        Assert.That(manager.CanProceed, Is.True);

        const BindingFlags instancePrivate = BindingFlags.Instance | BindingFlags.NonPublic;
        const BindingFlags staticPrivate = BindingFlags.Static | BindingFlags.NonPublic;
        Type managerType = typeof(GameSceneManager);
        MethodInfo getter = managerType.GetProperty("CanProceed").GetGetMethod();
        MethodInfo registered = managerType.GetMethod("GetRegisteredOwners", instancePrivate);
        MethodInfo prepared = managerType.GetMethod("IsOwnerPrepared", staticPrivate);
        MethodInfo expected = managerType.GetMethod("ExpectedGameScenes", instancePrivate);
        MethodInfo inventory = managerType.GetMethod("InventoryMatches", staticPrivate);
        if (registered == null || prepared == null || expected == null || inventory == null)
            throw new InvalidOperationException("Original private component signatures are absent; do not use this probe after optimization.");

        Type ownerType = prepared.GetParameters()[0].ParameterType;
        Type predicateType = typeof(Func<,>).MakeGenericType(ownerType, typeof(bool));
        ConstructorInfo predicateConstructor = predicateType.GetConstructor(new[] { typeof(object), typeof(IntPtr) });
        MethodInfo allDefinition = typeof(Enumerable).GetMethods(BindingFlags.Public | BindingFlags.Static)
            .Single(method => method.Name == "All" && method.IsGenericMethodDefinition && method.GetParameters().Length == 2);
        MethodInfo allMethod = allDefinition.MakeGenericMethod(ownerType);
        byte[] getterIl = getter.GetMethodBody().GetILAsByteArray();
        GetterIlHex = BitConverter.ToString(getterIl);
        OriginalAllSequenceVerified = VerifyAllSequence(getter, getterIl, registered, prepared, predicateConstructor, allMethod);
        if (!OriginalAllSequenceVerified)
        {
            return new
            {
                completed = false,
                reason = "Getter IL does not contain GetRegisteredOwners / ldnull / ldftn IsOwnerPrepared / new Func<Owner,bool> / Enumerable.All in order. A cached delegate or different compilation must be investigated; no component measurements performed.",
                getterIlHex = GetterIlHex,
                allSequenceVerified = false
            };
        }

        var allProxy = new DynamicMethod("Imp05OriginalAllSequence", typeof(bool),
            new[] { managerType }, managerType.Module, true);
        ILGenerator il = allProxy.GetILGenerator();
        il.Emit(OpCodes.Ldarg_0);
        il.Emit(OpCodes.Call, registered);
        il.Emit(OpCodes.Ldnull);
        il.Emit(OpCodes.Ldftn, prepared);
        il.Emit(OpCodes.Newobj, predicateConstructor);
        il.Emit(OpCodes.Call, allMethod);
        il.Emit(OpCodes.Ret);
        var readAll = (Func<GameSceneManager, bool>)allProxy.CreateDelegate(typeof(Func<GameSceneManager, bool>));
        var readExpected = (Func<Scene[]>)Delegate.CreateDelegate(typeof(Func<Scene[]>), manager, expected);
        var readInventory = (Func<Scene[], bool>)Delegate.CreateDelegate(typeof(Func<Scene[], bool>), inventory);
        var readWhole = (Func<bool>)Delegate.CreateDelegate(typeof(Func<bool>), manager, getter);

        // Reflection/Emit/binding, expected input preparation, and one component warm-up are outside windows.
        InventoryExpected = readExpected();
        Assert.That(readAll(manager), Is.True);
        ExpectedSink = readExpected();
        Assert.That(readInventory(InventoryExpected), Is.True);
        Assert.That(readWhole(), Is.True);

        string[] names = { "registered iterator + new predicate delegate + Enumerable.All proxy",
            "actual ExpectedGameScenes", "actual InventoryMatches with prebuilt expected array", "actual CanProceed getter" };
        Action[] workloads =
        {
            () => { bool value = true; for (int i = 0; i < CallsPerComponent; i++) value &= readAll(manager); BooleanSink = value; },
            () => { for (int i = 0; i < CallsPerComponent; i++) ExpectedSink = readExpected(); },
            () => { bool value = true; for (int i = 0; i < CallsPerComponent; i++) value &= readInventory(InventoryExpected); BooleanSink = value; },
            () => { bool value = true; for (int i = 0; i < CallsPerComponent; i++) value &= readWhole(); BooleanSink = value; }
        };
        for (int i = 0; i < workloads.Length; i++)
        {
            GcAllocationMeasurementResult measurement = GcAllocationMeasurement.Measure(workloads[i]);
            Results[i] = new ComponentResult
            {
                Component = names[i], Calls = CallsPerComponent, Events = measurement.EventCount,
                PositiveControlEvents = measurement.PositiveControlEventCount,
                NegativeControlEvents = measurement.NegativeControlEventCount,
                FrameBytesAvailable = measurement.FrameBytesAvailable,
                GlobalFrameByteDelta = measurement.FrameByteDelta
            };
            if (i != 1) Assert.That(BooleanSink, Is.True);
        }
        Assert.That(manager.State, Is.EqualTo(SceneTransitionState.Ready));
        Assert.That(manager.CanProceed, Is.True);
        return new
        {
            completed = true, allSequenceVerified = OriginalAllSequenceVerified, getterIlHex = GetterIlHex,
            observations = Results,
            limitations = "All is a DynamicMethod proxy reproducing the verified allocation-bearing IL sequence, not an independently extracted original compiled block. ExpectedGameScenes, InventoryMatches, and CanProceed are the actual bound methods. Component runs are separate windows on the same Ready instance; counts are not assumed additive. Event counts are not bytes; frame bytes are global diagnostics. Caller owns setup/cleanup and serialization outside windows. This original-symbol probe must not be reused after method replacement."
        };
    }

    private static bool SameMember(MethodBase actual, MethodBase expected)
    {
        return actual != null && expected != null && actual.Module == expected.Module &&
            actual.MetadataToken == expected.MetadataToken;
    }

    private static bool VerifyAllSequence(MethodInfo getter, byte[] bytes, MethodInfo registered,
        MethodInfo prepared, ConstructorInfo predicateConstructor, MethodInfo allMethod)
    {
        var codes = new Dictionary<short, OpCode>();
        foreach (FieldInfo field in typeof(OpCodes).GetFields(BindingFlags.Public | BindingFlags.Static))
        {
            if (field.FieldType == typeof(OpCode))
            {
                OpCode code = (OpCode)field.GetValue(null);
                codes[code.Value] = code;
            }
        }
        var instructions = new List<Instruction>();
        int cursor = 0;
        while (cursor < bytes.Length)
        {
            short key = bytes[cursor++];
            if (key == 0xFE) key = unchecked((short)(0xFE00 | bytes[cursor++]));
            if (!codes.TryGetValue(key, out OpCode code)) throw new InvalidOperationException("Unknown getter IL opcode.");
            MethodBase method = null;
            int operandBytes;
            switch (code.OperandType)
            {
                case OperandType.InlineNone: operandBytes = 0; break;
                case OperandType.ShortInlineBrTarget:
                case OperandType.ShortInlineI:
                case OperandType.ShortInlineVar: operandBytes = 1; break;
                case OperandType.InlineVar: operandBytes = 2; break;
                case OperandType.InlineI8:
                case OperandType.InlineR: operandBytes = 8; break;
                case OperandType.InlineSwitch: operandBytes = 4 + 4 * BitConverter.ToInt32(bytes, cursor); break;
                case OperandType.InlineMethod:
                    method = getter.Module.ResolveMethod(BitConverter.ToInt32(bytes, cursor));
                    operandBytes = 4;
                    break;
                default: operandBytes = 4; break;
            }
            cursor += operandBytes;
            if (cursor > bytes.Length) throw new InvalidOperationException("Malformed getter IL operand.");
            if (code != OpCodes.Nop) instructions.Add(new Instruction { Code = code, Method = method });
        }
        for (int i = 0; i + 4 < instructions.Count; i++)
        {
            if ((instructions[i].Code == OpCodes.Call || instructions[i].Code == OpCodes.Callvirt) &&
                SameMember(instructions[i].Method, registered) &&
                instructions[i + 1].Code == OpCodes.Ldnull &&
                instructions[i + 2].Code == OpCodes.Ldftn && SameMember(instructions[i + 2].Method, prepared) &&
                instructions[i + 3].Code == OpCodes.Newobj && SameMember(instructions[i + 3].Method, predicateConstructor) &&
                instructions[i + 4].Code == OpCodes.Call && SameMember(instructions[i + 4].Method, allMethod))
                return true;
        }
        return false;
    }
}
